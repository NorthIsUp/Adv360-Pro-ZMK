## Findings from the prototype (read before implementing)

- **`&stp` needs C1's effects to do anything visible.** Pressing it selects underglow effect 5 (`UNDERGLOW_EFFECT_BATTERY`) and releasing it selects effect 4 (`UNDERGLOW_EFFECT_KINESIS`). Both exist only after S2 lands C1. On bare `main`, `zmk_rgb_underglow_select_effect(5)` returns `-EINVAL`. The S4-T1 test therefore asserts which effect the behavior requests (a `LOG_DBG` line) and not the return code, so S4 and S2 can land in either order. The hardware check (Mod + `&stp STP_BAT` shows the battery level) needs S2.
- **The fallback-to-trans (`d93499bb`) does not change behaviour on `main`.** `main` already returns `1` (`ZMK_BEHAVIOR_TRANSPARENT`) for a binding whose behavior device does not exist, and `zmk_keymap_position_state_changed` moves on to the next layer when it sees `ret > 0`. The Kinesis lookup names `"TRANS"` and `"trans"` never match: the stock `&trans` device is called `transparent` (its node name). The only observable change is the log text `… and no trans found`. Per A1 the port is verbatim, and the test pins both the fall-through and the Kinesis log line. See Risks for what would happen if a device named `trans` ever existed.
- **`mouse_key_press` overlaps upstream #2950.** `main` already has `&mkp` metadata (`MB1`…`MB5`) and `display-name = "Mouse Key Press"`. The pin has Kinesis names (`Left click`, `Right click`, `Middle click`, `MB4 click`, `MB5 click`) and no `display-name`: Kinesis added it in `d1a0660e` and removed it again in `dfb23a16`. Kinesis wins, so S4-T3 renames the five params and deletes the `display-name` line. After that the dtsi is byte-identical to the pin (blob `fcb76664`), and Studio shows the behavior as `mouse_key_press`, its device-name fallback.
- **The `&stp` node name is `behavior_stp_indicators` (the Kinesis name), not K5's `stp_indicators`.** The device name is the Studio behavior identity. `CONFIG_ZMK_BEHAVIOR_LOCAL_ID_TYPE_CRC16` hashes it (`app/src/behavior.c:301`, `crc16_ansi(item->device->name, …)`), and the settings table keys on it. Keeping the Kinesis name keeps Clique-saved `&stp` bindings valid across the upgrade. The label (`&stp`) and compatible match K5. This is recorded as a contract correction.
- The upstream `app/tests/studio` and `app/tests/pointing` suites still pass after C8 (run in the prototype).

`app/src/behaviors/behavior_stp_indicators.c`. This is the Kinesis driver with three changes: the metadata sits under `CONFIG_ZMK_BEHAVIOR_METADATA`, which avoids an unused static when metadata is off; the no-op init is dropped, as upstream did in #2843; and one `LOG_DBG` per edge is added for the test:

```c
/*
 * Copyright (c) 2026 The ZMK Contributors
 *
 * SPDX-License-Identifier: MIT
 */

#define DT_DRV_COMPAT zmk_behavior_stp_indicators

#include <zephyr/device.h>
#include <drivers/behavior.h>
#include <zephyr/logging/log.h>

#include <dt-bindings/zmk/stp.h>
#include <zmk/behavior.h>
#include <zmk/rgb_underglow.h>

LOG_MODULE_DECLARE(zmk, CONFIG_ZMK_LOG_LEVEL);

#if DT_HAS_COMPAT_STATUS_OKAY(DT_DRV_COMPAT)

// rgb_underglow.c effect indices (C1): UNDERGLOW_EFFECT_KINESIS and UNDERGLOW_EFFECT_BATTERY.
#define STP_EFFECT_KINESIS 4
#define STP_EFFECT_BATTERY 5

#if IS_ENABLED(CONFIG_ZMK_BEHAVIOR_METADATA)

static const struct behavior_parameter_value_metadata param_values[] = {
    {
        .value = STP_BAT,
        .display_name = "Battery command",
        .type = BEHAVIOR_PARAMETER_VALUE_TYPE_VALUE,
    },
};

static const struct behavior_parameter_metadata_set param_metadata_set[] = {{
    .param1_values = param_values,
    .param1_values_len = ARRAY_SIZE(param_values),
}};

static const struct behavior_parameter_metadata metadata = {
    .sets_len = ARRAY_SIZE(param_metadata_set),
    .sets = param_metadata_set,
};

#endif // IS_ENABLED(CONFIG_ZMK_BEHAVIOR_METADATA)

static int on_keymap_binding_pressed(struct zmk_behavior_binding *binding,
                                     struct zmk_behavior_binding_event event) {
    switch (binding->param1) {
    case STP_BAT:
        LOG_DBG("position %d STP_BAT pressed, effect %d", event.position, STP_EFFECT_BATTERY);
        return zmk_rgb_underglow_select_effect(STP_EFFECT_BATTERY);
    }

    return -ENOTSUP;
}

static int on_keymap_binding_released(struct zmk_behavior_binding *binding,
                                      struct zmk_behavior_binding_event event) {
    switch (binding->param1) {
    case STP_BAT:
        LOG_DBG("position %d STP_BAT released, effect %d", event.position, STP_EFFECT_KINESIS);
        return zmk_rgb_underglow_select_effect(STP_EFFECT_KINESIS);
    }

    return ZMK_BEHAVIOR_OPAQUE;
}

static const struct behavior_driver_api behavior_stp_indicators_driver_api = {
    .binding_pressed = on_keymap_binding_pressed,
    .binding_released = on_keymap_binding_released,
    .locality = BEHAVIOR_LOCALITY_CENTRAL,
#if IS_ENABLED(CONFIG_ZMK_BEHAVIOR_METADATA)
    .parameter_metadata = &metadata,
#endif // IS_ENABLED(CONFIG_ZMK_BEHAVIOR_METADATA)
};

BEHAVIOR_DT_INST_DEFINE(0, NULL, NULL, NULL, NULL, POST_KERNEL, CONFIG_KERNEL_INIT_PRIORITY_DEFAULT,
                        &behavior_stp_indicators_driver_api);

#endif // DT_HAS_COMPAT_STATUS_OKAY(DT_DRV_COMPAT)
```

/*

 * SPDX-License-Identifier: MIT

#define DT_DRV_COMPAT zmk_behavior_dump_metadata

#include <zephyr/device.h>
#include <drivers/behavior.h>
#include <zephyr/logging/log.h>

#include <zmk/keymap.h>
#include <zmk/behavior.h>

LOG_MODULE_DECLARE(zmk, CONFIG_ZMK_LOG_LEVEL);

#if IS_ENABLED(CONFIG_ZMK_BEHAVIOR_METADATA) && DT_HAS_COMPAT_STATUS_OKAY(DT_DRV_COMPAT)

// Logs what ZMK Studio (Clique) reads over RPC: each behavior's display name and
// param1 values, then every layer name.
static int on_dump_metadata_binding_pressed(struct zmk_behavior_binding *binding,
                                            struct zmk_behavior_binding_event event) {
    STRUCT_SECTION_FOREACH(zmk_behavior_ref, item) {
        LOG_DBG("behavior %s display \"%s\"", item->device->name, item->metadata.display_name);

        struct behavior_parameter_metadata md = {0};
        if (behavior_get_parameter_metadata(item->device, &md) < 0) {
            continue;
        }
        for (size_t s = 0; s < md.sets_len; s++) {
            for (size_t v = 0; v < md.sets[s].param1_values_len; v++) {
                const struct behavior_parameter_value_metadata *pv = &md.sets[s].param1_values[v];
                LOG_DBG("param %s %d 0x%x \"%s\"", item->device->name, pv->type, pv->value,
                        pv->display_name);
            }
        }
    }

    for (zmk_keymap_layer_id_t id = 0; id < ZMK_KEYMAP_LAYERS_LEN; id++) {
        LOG_DBG("layer %d name \"%s\"", id, zmk_keymap_layer_name(id));
    }

    return ZMK_BEHAVIOR_OPAQUE;
}

static int on_dump_metadata_binding_released(struct zmk_behavior_binding *binding,
                                             struct zmk_behavior_binding_event event) {
    return ZMK_BEHAVIOR_OPAQUE;
}

static const struct behavior_driver_api behavior_dump_metadata_driver_api = {
    .binding_pressed = on_dump_metadata_binding_pressed,
    .binding_released = on_dump_metadata_binding_released,
};

BEHAVIOR_DT_INST_DEFINE(0, NULL, NULL, NULL, NULL, POST_KERNEL, CONFIG_KERNEL_INIT_PRIORITY_DEFAULT,
                        &behavior_dump_metadata_driver_api);

#endif // IS_ENABLED(CONFIG_ZMK_BEHAVIOR_METADATA) && DT_HAS_COMPAT_STATUS_OKAY(DT_DRV_COMPAT)
